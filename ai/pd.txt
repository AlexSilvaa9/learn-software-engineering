# Acceder a la primera fila (por posición)
print(df.iloc[0])

# Acceder a las primeras 2 filas
print(df.iloc[:2])

# Acceder a una fila específica por posición y columna
print(df.iloc[0, 1])  # Primera fila, segunda columna (Edad de Ana)


# Acceder a una fila por nombre de índice
print(df.loc['Ana'])

# Acceder a una fila por nombre y columna
print(df.loc['Ana', 'Edad'])  # Edad de Ana



# Filtrar varias condiciones (por ejemplo, Edad > 25 y Ciudad == 'Madrid')
print(df[(df['Edad'] > 25) & (df['Ciudad'] == 'Madrid')])



# Filtrar con query
print(df.query('Edad > 25 and Ciudad == "Madrid"'))

# Acceder a múltiples columnas
print(df[['Edad', 'Ciudad']])

# Obtener un valor específico (por posición)
print(df.iloc[1, 2])  # Segunda fila, tercera columna (Valor de 'Ciudad')

# Obtener un valor específico (por nombre de índice)
print(df.loc['Luis', 'Edad'])  # Edad de 'Luis'

# Seleccionar las primeras dos columnas por su posición
print(df.iloc[:, 0:2])
